// Native status-line decision and geometry probe, assembled by the Python builder.
//
// It exercises draw_status_info() across widths, flow states and warning conditions
// and checks what makes the status line a contract rather than a drawing:
//
//   1. the metric form is the richest that fits, and no richer;
//   2. the version is drawn only when the chosen content plus it fits, and it ends
//      exactly on the right margin of the line;
//   3. the warnings of GitHub issue #17 §4 are never dropped, at any width;
//   4. a line whose content fits never draws past the right margin.
//
// The oracle re-applies the policy from the same primitives (status_build,
// status_measure), which is the point: the policy lives in one place.
//
// A second line was implemented and measured in 0.10.0 and removed in 0.11.0: it
// cost two rows of pairs on a 60-pair session and its trigger made the choice
// bistable around the row capacity. This probe is the one-line contract.
//
// gmem namespace CCScopeStatusTest: 20 status (0 running, 1 passed, -1 failed),
// 21 completed case, 22 failing case, 23 expected form, 24 actual form,
// 27 failing check.

probe_case = 0;
probe_cases = 180;
gmem[20] = 0;
gmem[21] = 0;

// Which assertion failed: a probe that says "case 105" without saying why costs a
// round trip through REAPER to diagnose.
function probe_fail(code) (
  gmem[20] = -1;
  gmem[22] = probe_case;
  gmem[27] = code;
  0;
);

// Nine widths, from "nothing fits" to wider than any window this plugin is used in.
function probe_width(i) (
  i == 0 ? 180 : i == 1 ? 220 : i == 2 ? 260 : i == 3 ? 300 : i == 4 ? 360 :
  i == 5 ? 420 : i == 6 ? 520 : i == 7 ? 700 : 1016;
);

// Five states: Waiting, Receiving with one stream, Receiving with four, Idle, Frozen.
function probe_apply_state(state) local() (
  slider1 = 0;
  slider6 = 0;
  slider2 = 0;
  slider7 = 0;
  g_sel_streams = 0;
  g_sel_channels = 0;
  g_sel_msg_s = 0;
  probe_count = 0;

  state == 0 ? (
    probe_count = 0;                       // Waiting: the selection holds no pair with data
  ) : state == 1 ? (
    probe_count = 2;                       // Receiving: one stream on one channel
    g_sel_streams = 1;
    g_sel_channels = 1;
    g_sel_msg_s = 20;
  ) : state == 2 ? (
    probe_count = 7;                       // Receiving: several streams, plural text
    g_sel_streams = 4;
    g_sel_channels = 3;
    g_sel_msg_s = 240;
  ) : state == 3 ? (
    probe_count = 5;                       // Idle: data exists, nothing in the last second
  ) : (
    probe_count = 5;                       // Frozen: the parameter wins over the rest
    slider2 = 1;
  );
);

function probe_apply_warning(warn) (
  g_visible_bars = 40;
  g_side_fallback = 0;
  warn == 1 || warn == 3 ? ( g_visible_bars = 2; );
  warn == 2 || warn == 3 ? ( g_side_fallback = 1; );
);

// The policy, re-derived: the richest form whose CONTENT fits, because the version
// is the first thing to go -- never a reason to shorten the numbers.
function probe_form_that_fits(avail) local(w3, w2) (
  status_build(probe_count, 3);
  w3 = status_measure();
  status_build(probe_count, 2);
  w2 = status_measure();
  w3 <= avail ? 3 : w2 <= avail ? 2 : 1;
);

function probe_version_fits(form, avail) local() (
  status_build(probe_count, form);
  status_measure() + g_st_ver_w + 8 <= avail;
);

@gfx 1000 720
gfx_clear = 0;
gfx_mode = 0;
gfx_setfont(1, "Arial", FONT_SIZE);
gfx_h = 720;

// The widths the oracle needs are measured by draw_status_info() itself, on its
// first call. Without that call the oracle would treat the version as zero pixels
// wide and demand it in windows that cannot hold it.
g_st_ready == 0 ? ( draw_status_info(0); );

probe_case < probe_cases && gmem[20] != -1 ? (
  gfx_w = probe_width(probe_case % 9);
  probe_state = floor(probe_case / 9) % 5;
  probe_warn = floor(probe_case / 45) % 4;
  probe_apply_state(probe_state);
  probe_apply_warning(probe_warn);

  probe_avail = gfx_w - 20;
  draw_status_info(probe_count);

  // Snapshot what the implementation decided before asking the oracle anything: the
  // oracle rebuilds the pieces, and a flag read afterwards would be the oracle's.
  probe_got_form = g_st_form;
  probe_got_ver = g_st_ver;
  probe_got_wv = g_st_has_wv;
  probe_got_ws = g_st_has_ws;
  probe_got_w = g_st_w;
  probe_got_end_x = gfx_x;
  probe_got_end_y = gfx_y;
  probe_got_y2 = g_st_y2;

  // 1. the metric detail: only Receiving has one to sacrifice
  probe_expected_form = (probe_state == 1 || probe_state == 2) ? probe_form_that_fits(probe_avail) : 3;
  (probe_state == 1 || probe_state == 2) && probe_got_form != probe_expected_form ? (
    gmem[23] = probe_expected_form;
    gmem[24] = probe_got_form;
    probe_fail(4);
  );

  // 2. the version, in what is left after the content has been decided
  probe_expected_ver = probe_version_fits(probe_expected_form, probe_avail) ? 1 : 0;
  probe_got_ver != probe_expected_ver ? ( probe_fail(5); );

  // 3. the warnings are the reason the line exists: they never disappear
  probe_count > g_visible_bars && !probe_got_wv ? ( probe_fail(6); );
  g_side_fallback && !probe_got_ws ? ( probe_fail(7); );

  // 4. geometry: content that fits stays inside the margin, and a drawn version
  //    ends on it, on the line's baseline
  probe_got_w <= probe_avail ? ( probe_got_end_x > gfx_w - 10 + 1 ? ( probe_fail(8); ); );
  probe_got_ver ? (
    abs(probe_got_end_x - (gfx_w - 10)) > 2 ? ( probe_fail(9); );
    probe_got_end_y != probe_got_y2 ? ( probe_fail(10); );
  );

  probe_case += 1;
  gmem[21] = probe_case;
  probe_case == probe_cases && gmem[20] != -1 ? ( gmem[20] = 1; );
);

gfx_r = gmem[20] == -1 ? 1 : 0.2;
gfx_g = gmem[20] == -1 ? 0.2 : 0.8;
gfx_b = 0.2;
gfx_a = 1;
gfx_x = 20;
gfx_y = 20;
gfx_printf("Status line: %d / %d cases, status %d", probe_case, probe_cases, gmem[20]);
gmem[20] == -1 ? (
  gfx_x = 20;
  gfx_y = 44;
  gfx_printf("failed case %d check %d: form %d (expected %d)", gmem[22], gmem[27], gmem[24], gmem[23]);
);
